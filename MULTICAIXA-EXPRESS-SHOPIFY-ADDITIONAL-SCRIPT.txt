<!-- ##### GET paymentMethod and paymentAmount ##### -->
	{% assign paymentMethod = '' %}
	{% for transaction in transactions %}
		{% assign gateway = transaction.gateway | upcase %}
        {% if gateway | strip == "MULTICAIXA EXPRESS" %}
            {% assign paymentMethod = gateway %}
            {% assign paymentAmount  = transaction.amount | money_without_currency %} 
		{% endif %}
	{% endfor %}
<!-- ##### END OF GET paymentMethod and paymentAmount ##### -->

<!-- ##### FORMAT paymentAmount for decimal dot ##### -->
    {% assign count = 0 %}
    {% assign result = '' %}
    {% assign num = paymentAmount | size | minus: 1 %}
    {% assign range = (0..num) %}

    {% for i in range %}
        {% assign paymentAmountChar = paymentAmount | slice: i %}
        {% if paymentAmountChar == "." or paymentAmountChar == "," %}
            {% assign count  = count | plus: 1 %}
        {% endif %}
    {% endfor %}

    {% if count > 1 %}
        {% for i in range %}
            {% assign paymentAmountChar = paymentAmount | slice: i %}      
            {% if count > 1 %}
                {% if paymentAmountChar == "." or paymentAmountChar == "," %}
                    {% assign count  = count | minus: 1 %}
                {% else %}
                    {% assign result = result | append: paymentAmountChar %}
                {% endif %}
            {% else %}
                {% assign result = result | append: paymentAmountChar %}
            {% endif %}
        {% endfor %}
    {% else %}
        {% assign result = paymentAmount %}
    {% endif %}

    {% assign result = result | replace: ",", "." %}
<!-- ##### END OF FORMAT paymentAmount for decimal dot ##### -->
